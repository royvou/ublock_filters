! Paid priority badges. Match badge text, never listing titles/descriptions.
! Both exclusions share one candidate scan. Remove either regex alternative to keep that type.
marktplaats.nl##:is(.hz-Listing,.hz-StructuredListing) :is(.hz-Listing-priority,.hz-Listing-priority-new,[class*="hz-Listing-priority-new"]):has-text(/^\s*(?:Topadvertentie|Dagtopper)\s*$/i):upward(.hz-Listing,.hz-StructuredListing)

! Admarkt cards, including those without a website button.
! Match an Admarkt ID in the URL path; do not match /a in category names.
! Native /a prefilter reduces regex candidates; it is not enough to identify an ad by itself.
! Inspect the primary listing link, not links in same-seller recommendations.
marktplaats.nl##.hz-Listing a:is(.hz-Listing-coverLink,[class*="hz-Listing-coverLink"])[href*="/v/"][href*="/a" i]:matches-attr(href=/^[^?#]*\/a\d+(?:[-/]|[?#]|$)/i):upward(.hz-Listing)
marktplaats.nl##.hz-StructuredListing a[href*="/a" i]:matches-attr(href=/^[^?#]*\/a\d+(?:[-/]|[?#]|$)/i):upward(.hz-StructuredListing)
! A seller's website link alone does not identify a promoted listing.

! Promotional stickers: hide the card, not just its sticker.
marktplaats.nl##.hz-Listing:has([data-testid="listing-opval-sticker"],.hz-Listing-Opvalsticker-wrapper,.hz-Listing-Opvalsticker-wrapper-new)

! Homepage advertisement tiles.
marktplaats.nl##.hz-Listing-imageOverlayLabel:has-text(/^\s*Homepagina-advertentie\s*$/i):upward(.hz-Link.hz-Link--block)
